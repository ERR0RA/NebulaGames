const canvas = document.getElementById('court'); const ctx = canvas.getContext('2d'); canvas.width = 1000; canvas.height = 600; let gameActive = false; let screenShake = 0; const particles = []; const stars = Array.from({length: 100}, () => ({x: Math.random()*1000, y: Math.random()*600, s: Math.random()*2})); const gravity = 0.45; const friction = 0.98; const bounciness = 0.8; const FLOOR_Y = 540; let score1 = 0, score2 = 0; const keys = {}; const ball = { x: 500, y: 150, vx: 5, vy: 0, r: 14, trail: [], holder: null, // Track who is holding the ball reset(dir) { this.x = 500; this.y = 150; this.vx = dir * 7; this.vy = -5; this.trail = []; this.holder = null; } }; class Player { constructor(x, color, team, controlType) { this.x = x; this.y = 400; this.vx = 0; this.vy = 0; this.w = 45; this.h = 90; this.color = color; this.team = team; this.controlType = controlType; this.grounded = false; this.moveSpeed = 1.3; this.maxSpeed = 9; this.jumpForce = -15; this.animTimer = 0; } update() { let left, right, jump; if (this.controlType === 'p1') { left = keys['KeyA'] || keys['ArrowLeft']; right = keys['KeyD'] || keys['ArrowRight']; jump = keys['KeyW'] || keys['ArrowUp'] || keys['Space']; } else if (this.controlType === 'p2') { left = keys['KeyJ']; right = keys['KeyL']; jump = keys['KeyI']; } else if (this.controlType === 'bot') { let targetX = ball.x - (this.team === 1 ? -40 : 40); this.vx += Math.sign(targetX - this.x) * 0.75; if (Math.abs(ball.x - this.x) < 100 && ball.y < this.y - 20 && this.grounded) { this.vy = this.jumpForce; } } if (this.controlType !== 'bot') { if (left) this.vx -= this.moveSpeed; else if (right) this.vx += this.moveSpeed; else this.vx *= 0.82; if (jump && this.grounded) { this.vy = this.jumpForce; this.grounded = false; screenShake = 4; } } this.vx = Math.max(-this.maxSpeed, Math.min(this.maxSpeed, this.vx)); this.vy += gravity; this.x += this.vx; this.y += this.vy; if (this.y + this.h > FLOOR_Y) { this.y = FLOOR_Y - this.h; this.vy = 0; this.grounded = true; } if (this.x < 0) this.x = 0; if (this.x + this.w > canvas.width) this.x = canvas.width - this.w; // Ball Collision & Grabbing if (ball.holder === null) { let cx = Math.max(this.x, Math.min(ball.x, this.x + this.w)); let cy = Math.max(this.y, Math.min(ball.y, this.y + this.h)); let dx = ball.x - cx, dy = ball.y - cy; let dist = Math.sqrt(dx*dx + dy*dy); if (dist < ball.r + 5) { // If moving fast downwards or jumping high, grab the ball for a dunk if (!this.grounded && Math.abs(this.vy) > 2) { ball.holder = this; } else { // Regular bounce let nx = dx / (dist || 1), ny = dy / (dist || 1); ball.vx = nx * 12 + this.vx; ball.vy = ny * 12 + (this.vy < 0 ? this.vy : -4); spawnParticles(ball.x, ball.y, "#fff", 5); } } } else if (ball.holder === this) { // Ball follows player ball.x = this.x + this.w/2 + (this.team === 1 ? 20 : -20); ball.y = this.y + 10; ball.vx = this.vx; ball.vy = this.vy; // Slam Dunk Detection let nearHoop = (this.team === 1 && ball.x > 900 && ball.y < 300) || (this.team === 2 && ball.x < 100 && ball.y < 300); if (nearHoop) { this.performDunk(); } // Release ball if grounded if (this.grounded) ball.holder = null; } } performDunk() { const isTeam1 = this.team === 1; if (isTeam1) { score1++; resetRound('s1', score1, -1, "SLAM DUNK!"); } else { score2++; resetRound('s2', score2, 1, "SLAM DUNK!"); } screenShake = 30; spawnParticles(ball.x, ball.y, this.color, 50); ball.holder = null; } draw() { ctx.save(); ctx.translate(this.x + this.w/2, this.y + this.h/2); // Lean forward when running or dunking let tilt = ball.holder === this ? (this.team === 1 ? 0.4 : -0.4) : this.vx * 0.03; ctx.rotate(tilt); ctx.shadowBlur = 20; ctx.shadowColor = this.color; ctx.fillStyle = this.color; ctx.fillRect(-this.w/2, -15, this.w, 40); ctx.fillStyle = "#ffdbac"; ctx.beginPath(); ctx.arc(0, -35, 18, 0, Math.PI*2); ctx.fill(); ctx.fillStyle = "#222"; ctx.fillRect(-this.w/2, 25, 15, 20); ctx.fillRect(this.w/2 - 15, 25, 15, 20); ctx.fillStyle = "white"; ctx.font = "bold 15px Arial"; ctx.textAlign = "center"; ctx.fillText(this.controlType.toUpperCase(), 0, 10); ctx.restore(); } } let t1, t2; function startGame(mode) { document.getElementById('menu').style.display = 'none'; document.getElementById('ui').style.display = 'flex'; canvas.style.display = 'block'; if (mode === 'P1_VS_BOT') { t1 = new Player(150, '#ff3e3e', 1, 'p1'); t2 = new Player(800, '#00d2ff', 2, 'bot'); } else if (mode === 'P1_VS_P2') { t1 = new Player(150, '#ff3e3e', 1, 'p1'); t2 = new Player(800, '#00d2ff', 2, 'p2'); } else if (mode === 'BOT_VS_BOT') { t1 = new Player(150, '#ff3e3e', 1, 'bot'); t2 = new Player(800, '#00d2ff', 2, 'bot'); } gameActive = true; loop(); } window.onkeydown = e => keys[e.code] = true; window.onkeyup = e => keys[e.code] = false; function spawnParticles(x, y, color, count = 12) { for (let i = 0; i < count; i++) { particles.push({ x, y, vx: (Math.random()-0.5)*15, vy: (Math.random()-0.5)*15, alpha: 1, color, r: Math.random()*3+1 }); } } function drawBackground() { let grad = ctx.createRadialGradient(500, 300, 50, 500, 300, 600); grad.addColorStop(0, "#0a0a25"); grad.addColorStop(1, "#020205"); ctx.fillStyle = grad; ctx.fillRect(0, 0, canvas.width, canvas.height); ctx.fillStyle = "white"; stars.forEach(s => ctx.fillRect(s.x, s.y, s.s, s.s)); ctx.fillStyle = "#050510"; ctx.fillRect(0, FLOOR_Y, canvas.width, 60); ctx.strokeStyle = "#4444ff"; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo(0, FLOOR_Y); ctx.lineTo(canvas.width, FLOOR_Y); ctx.stroke(); } function loop() { if (!gameActive) return; let sx = (Math.random()-0.5)*screenShake; let sy = (Math.random()-0.5)*screenShake; ctx.setTransform(1, 0, 0, 1, sx, sy); screenShake *= 0.9; drawBackground(); // Platform ctx.fillStyle = "rgba(255,255,255,0.1)"; ctx.strokeStyle = "#4deeea"; ctx.strokeRect(400, 320, 200, 10); ctx.fillRect(400, 320, 200, 10); t1.update(); t2.update(); // Ball Physics (only if not held) if (!ball.holder) { ball.vy += gravity; ball.vx *= friction; ball.x += ball.vx; ball.y += ball.vy; if (ball.x < ball.r || ball.x > canvas.width - ball.r) { ball.vx *= -bounciness; ball.x = ball.x < ball.r ? ball.r : canvas.width - ball.r; } if (ball.y < ball.r) { ball.vy *= -bounciness; ball.y = ball.r; } if (ball.y + ball.r > FLOOR_Y) { ball.y = FLOOR_Y - ball.r; ball.vy *= -bounciness; } // Platform bounce if (ball.x > 400 && ball.x < 600 && ball.y + ball.r > 320 && ball.y - ball.r < 335) { ball.y = ball.vy > 0 ? 320 - ball.r : 335 + ball.r; ball.vy *= -bounciness; } // Normal Goal Detection if (ball.y > 240 && ball.y < 265) { if (ball.x < 50) { score2++; resetRound('s2', score2, 1, "GOAL!"); } if (ball.x > 950) { score1++; resetRound('s1', score1, -1, "GOAL!"); } } } // Ball Trail ball.trail.push({x: ball.x, y: ball.y}); if (ball.trail.length > 12) ball.trail.shift(); ball.trail.forEach((p, i) => { ctx.beginPath(); ctx.arc(p.x, p.y, ball.r * (i/12), 0, Math.PI*2); ctx.fillStyle = ball.holder ? `rgba(0, 255, 255, ${i/20})` : `rgba(255, 165, 0, ${i/20})`; ctx.fill(); }); particles.forEach((p, i) => { p.x += p.vx; p.y += p.vy; p.alpha -= 0.02; ctx.globalAlpha = p.alpha; ctx.fillStyle = p.color; ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI*2); ctx.fill(); if (p.alpha <= 0) particles.splice(i, 1); }); ctx.globalAlpha = 1; // Draw Hoops ctx.lineWidth = 6; ctx.strokeStyle = "#ff3e3e"; ctx.strokeRect(0, 250, 45, 8); ctx.strokeStyle = "#00d2ff"; ctx.strokeRect(955, 250, 45, 8); t1.draw(); t2.draw(); // Draw Ball ctx.shadowBlur = ball.holder ? 40 : 25; ctx.shadowColor = ball.holder ? "#00ffff" : "#ffaa00"; ctx.fillStyle = ball.holder ? "#00ffff" : "#ffaa00"; ctx.beginPath(); ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI*2); ctx.fill(); ctx.shadowBlur = 0; requestAnimationFrame(loop); } function resetRound(id, val, dir, text) { document.getElementById(id).innerText = val; const msg = document.getElementById('msg'); msg.innerText = text; msg.style.display = 'block'; gameActive = false; // Pause physics setTimeout(() => { msg.style.display = 'none'; ball.reset(dir); gameActive = true; }, 1200); }
