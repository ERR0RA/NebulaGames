update() { if (!this.alive) return; if (this.type === 'mouse') { let dx = mouse.x - this.x; let dy = mouse.y - this.y; let dist = Math.sqrt(dx*dx + dy*dy); if (dist > 5) { this.vx += (dx / dist) * this.speed; this.vy += (dy / dist) * this.speed; } } else if (this.type === 'keys') { if (keys['ArrowUp'] || keys['KeyW']) this.vy -= this.speed; if (keys['ArrowDown'] || keys['KeyS']) this.vy += this.speed; if (keys['ArrowLeft'] || keys['KeyA']) this.vx -= this.speed; if (keys['ArrowRight'] || keys['KeyD']) this.vx += this.speed; } else if (this.type === 'bot') { let target = players.find(p => p !== this && p.alive && p.team !== this.team); if (target) { // Added jitter so they don't aim perfectly at the center of the other orb const time = Date.now() * 0.002; let targetOffsetX = Math.sin(time + this.id) * 40; let targetOffsetY = Math.cos(time + this.id) * 40; let dx = (target.x + targetOffsetX) - this.x; let dy = (target.y + targetOffsetY) - this.y; let dist = Math.sqrt(dx*dx + dy*dy); // Behavior: If very close, push harder; if far, approach cautiously let intelligence = dist < 100 ? this.speed * 1.2 : this.speed * 0.8; // Arena Awareness: If near the edge, try to move toward the center instead of chasing let distFromCenter = Math.sqrt((this.x - canvas.width/2)**2 + (this.y - canvas.height/2)**2); if (distFromCenter > arenaRadius * 0.7) { dx = (canvas.width/2) - this.x; dy = (canvas.height/2) - this.y; intelligence = this.speed * 0.5; } this.vx += (dx / dist) * intelligence; this.vy += (dy / dist) * intelligence; } } this.vx *= this.friction; this.vy *= this.friction; this.x += this.vx; this.y += this.vy; let distFromCenter = Math.sqrt((this.x - canvas.width/2)**2 + (this.y - canvas.height/2)**2); if (distFromCenter > arenaRadius) { this.alive = false; createExplosion(this.x, this.y, this.color); checkRoundOver(); } }
